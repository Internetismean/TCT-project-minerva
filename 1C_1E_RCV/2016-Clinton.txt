//Auxiliary function that create a table full of 0s
function createMatrix(n) {
    return Array.from({ length: n }, () =>
        Array(n).fill(0)
    );
}

//Auxiliary function to calculate the similarity of two arrays
function similarity(A, B) {
    let dist = 0;
    for (let k = 0; k < A.length; k++) {
        dist += Math.abs(A[k] - B[k]);
    }
    return 1 - dist / (2 * A.length);
}

//Auxiliary function to calculate what percentage of voters will follow an endorsement
function endorsementFollowRate(posA, posB, statePos, gamma = 1.5) {
    const simAB = similarity(posA, posB);
    const simBS = similarity(posB, statePos);
    const simAS = similarity(posA, statePos);

    const rawAffinity = simAB * simBS * simAS;
    return Math.pow(rawAffinity, gamma);
}

//Auxiliary function to set starting values for matrixes
function computeTransferMatrix(candidates, candidatePositions, statePosition) {
    const N = candidates.length;
    const matrix = Array.from({ length: N }, () => Array(N).fill(0));

    for (let i = 0; i < N; i++) {
        const A = candidates[i];
        const posA = candidatePositions.get(A);

        let attraction = Array(N).fill(0);
        let totalPull = 0;

        for (let j = 0; j < N; j++) {
            if (i === j) continue;

            const B = candidates[j];
            const posB = candidatePositions.get(B);

            const simAB = similarity(posA, posB);
            const simBS = similarity(posB, statePosition);

            const score = simAB * simBS;
            attraction[j] = score;
            totalPull += score;
        }

        // Diagonal = abstention
        const abstention = totalPull > 0 ? 1 / (1 + totalPull) : 1;
        matrix[i][i] = abstention;

        // Off-diagonal = proportional redistribution
        if (totalPull > 0) {
            const transferable = 1 - abstention;
            for (let j = 0; j < N; j++) {
                if (i !== j) {
                    matrix[i][j] = transferable * (attraction[j] / totalPull);
                }
            }
        }
    }
    return matrix;
}

//givingCandidate and receivingCandidate are the ID for tables, not the pk.
function oneWayEndorsement(givingCandidate, receivingCandidate)
{
    for (const pk of statesPKs) 
    {
        var state_matrix = state_matrixes.get(pk);
        var receivingCandidatePk = candidateMap.get(receivingCandidate);
        for (let i = 0; i < state_matrix.length; i++)
        {
            if (i != receivingCandidate)
            {
                var comparingCandidatePk = candidateMap.get(i);
                var affinity = endorsementFollowRate(candidateIssueScores.get(comparingCandidatePk), candidateIssueScores.get(receivingCandidatePk), stateIssueScores.get(pk));
                var percentChange = state_matrix[givingCandidate][i] * affinity;
                state_matrix[givingCandidate][i] -= percentChange;
                state_matrix[givingCandidate][receivingCandidate] += percentChange;
            }
        }
    }
}

//candidateA and candidateB are the ID for tables, not the pk.
function coEndorsement(candidateA, candidateB)
{
    oneWayEndorsement(candidateA, candidateB);
    oneWayEndorsement(candidateB, candidateA);
}

//candidateList is a list of IDs for tables, not pks. orgScores are the position of the organization on issues, like if it were a candidate or state
function orgEndorsement(candidateList, orgScores, gamma = 1.2) {
    for (const statePk of statesPKs) {
        const matrix = state_matrixes.get(statePk);
        const stateScores = stateIssueScores.get(statePk);
        const N = matrix.length;

        for (let i = 0; i < N; i++) {
            if (dropOutCandidates.includes(i)) continue;
            const candidatePk = candidatesPKs[i];
            const voterPos = candidateIssueScores.get(candidatePk);

            if (!voterPos) continue;

            // How persuadable this bloc is by the org
            let influence = similarity(voterPos, orgScores) * similarity(orgScores, stateScores);

            influence = Math.max(0, Math.min(1, influence));

            if (influence === 0) continue;

            // Current abstention / loyalty
            const stay = matrix[i][i];
            const transferable = 1 - stay;

            if (transferable <= 0) continue;

            let attraction = {};
            let totalAttraction = 0;

            for (const endorsedID of candidateList) {
                if (endorsedID < 0 || endorsedID >= N || endorsedID === i) continue;

                const endorsedPk = candidateMap.get(endorsedID);
                const endorsedPos = candidateIssueScores.get(endorsedPk);
                if (!endorsedPos) continue;

                const score =
                    similarity(voterPos, endorsedPos) *
                    similarity(endorsedPos, orgScores) *
                    similarity(endorsedPos, stateScores);

                const weighted = Math.pow(Math.max(0, score), gamma);
                attraction[endorsedID] = weighted;
                totalAttraction += weighted;
            }

            if (totalAttraction === 0) continue;

            const shift = transferable * influence;

            // Reduce diagonal safely
            matrix[i][i] = stay - shift;

            // Redistribute
            for (const endorsedID in attraction) {
                const portion = attraction[endorsedID] / totalAttraction;
                matrix[i][endorsedID] += shift * portion;
            }
        }
    }
}

//Candidate is the ID for tables, not the pk
function candidateDropOut(candidate)
{
    for (state of e.states_json)
    {
        var state_multipliers = e.candidate_state_multiplier_json.filter( m => m.fields.state === state.pk);

        var loser_multiplier = state_multipliers.find(m => m.fields.candidate === candidateMap.get(candidate));
        
        var stateMatrix = state_matrixes.get(state.pk);
        var matrixLine = stateMatrix[candidate];

        for (index in matrixLine)
        {
            if (index != candidate)
            {
                var winner_multiplier = state_multipliers.find(m => m.fields.candidate === candidateMap.get(parseInt(index)));
                winner_multiplier.fields.state_multiplier = winner_multiplier.fields.state_multiplier + loser_multiplier.fields.state_multiplier * matrixLine[index];
            }
        }
        loser_multiplier.fields.state_multiplier = 0.0;
    }
    
    for (const pk of statesPKs)
    {
        const stateTable = state_matrixes.get(pk);
        stateTable[candidate].fill(0);
        for (let i = 0; i < stateTable.length; i++) {
            stateTable[i][candidate] = 0;
        }
        //Renormalize rows
        for (let i = 0; i < stateTable.length; i++) {
            const rowSum = stateTable[i].reduce((a, b) => a + b, 0);
            if (rowSum === 0) continue;

            const factor = 1 / rowSum;
            for (let j = 0; j < stateTable[i].length; j++) {
            stateTable[i][j] *= factor;
            }
        }
    }
    
    eliminatedCandidates.push(candidate);
    dropOutCandidates.push(candidate);
}

//receiver is the ID for tables, not the pk. givers is an array of table IDs for all candidates that will lose support. States is an array of 
//all state pks you want the effect to apply in. Percent is the quantity of support change (from 0 to 1 - e.g 0.05 is 5% increase and -0.05 is 5% decrease)
function candidateChangeSupportFromCandidate(receiver, givers, states, percent)
{
    for (const statePK of states)
    {
        var stateMatrix = state_matrixes.get(statePK);
        for (const loserID of givers)
        {
            var othersSum = 1 - stateMatrix[loserID][receiver];
            var ratio = percent / othersSum;
            for (var i = 0; i < stateMatrix.length; i++)
            {
                if (i != receiver)
                {
                    stateMatrix[loserID][i] -= ratio * stateMatrix[loserID][i];
                }
            }
            stateMatrix[loserID][receiver] += percent;
        }
    }
}

class RCVRound {
  constructor(candidateVoteDictionary, eliminatedCandidatesList, totalVotes) {
    this.candidateVoteDictionary = candidateVoteDictionary; //Dictionary candidate table IDs : total votes in the round;
    this.eliminatedCandidatesList = eliminatedCandidatesList; //List of eliminatedCandidates this round;
    this.totalVotes = totalVotes; //total valid votes this round;
  }
}

function updateStateResults(eliminatedCandidates)
{
    for (const pk of statesPKs)
    {
        var stateMatrix = state_matrixes.get(pk);
        var currentStateResult = e.final_state_results.find(s => s.state === pk).result;
        for (const ID of eliminatedCandidates)
        {
            var eliminatedCandidateVotes = currentStateResult.find(r => r.candidate === candidateMap.get(ID)).votes;
            for (var i = 0; i < stateMatrix.length; i++)
            {
                if (i != ID)
                {
                    currentStateResult.find(r => r.candidate === candidateMap.get(i)).votes += Math.round(eliminatedCandidateVotes * stateMatrix[ID][i]);
                }
            }
            stateMatrix[ID].fill(0);
            for (let i = 0; i < stateMatrix.length; i++) {
                stateMatrix[i][ID] = 0;
            }
            //Renormalize rows
            for (let i = 0; i < stateMatrix.length; i++) {
                const rowSum = stateMatrix[i].reduce((a, b) => a + b, 0);
                if (rowSum === 0) continue;

                const factor = 1 / rowSum;
                for (let j = 0; j < stateMatrix[i].length; j++) {
                stateMatrix[i][j] *= factor;
                }
            }
        }
    }
}
